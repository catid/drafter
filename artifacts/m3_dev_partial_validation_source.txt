"""Read-only bounded audit of the first eight completed real M3 standard dev bundles.

Uses the Python standard library only. It reads immutable result files and captures
the mutable manifest/session snapshots used. It never imports the experiment code,
loads tensors, initializes CUDA, changes source, or controls the running job.

checks = collections.Counter()
evidence = {}


def sha(data):
    return hashlib.sha256(data).hexdigest()


def fingerprint(value):
    return sha(json.dumps(value, sort_keys=True, separators=(',', ':')).encode())


def file_sha(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def read(path, mutable=False):
    data = path.read_bytes()
    evidence[str(path.relative_to(ROOT))] = {
        'sha256': sha(data), 'bytes': len(data), 'mutable_while_run_active': mutable}
    return json.loads(data)


def require(condition, label):
    if not condition:
        raise AssertionError(label)
    checks[label] += 1


def close(actual, expected):
    return math.isclose(actual, expected, rel_tol=1e-12, abs_tol=1e-12)


def finite(value):
    if isinstance(value, float):
        require(math.isfinite(value), 'all_recorded_numeric_values_finite')
    elif isinstance(value, dict):
        for child in value.values():
            finite(child)
    elif isinstance(value, list):
        for child in value:
            finite(child)

        aliases = {'prompt_id': 'prompt', 'arm': 'proposal_mode', 'settings': 'setting', 'checkpoint_hashes': 'checkpoint'}

                      'model_pair', 'model_revisions', 'stage', 'sampling_warp_config', 'timing_session', 'repeat', 'phase_profile'):
            require(row[field] == identity[field], 'raw_metadata_matches_identity')

        require(identity['model_pair'] == cfg['model_pair'] and identity['model_revisions'] == cfg['model_revisions'], 'raw_session_model_binding')

        require(identity['timing_session'] == session['logical_session'], 'raw_logical_session_binding')
