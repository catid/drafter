def sha(data):
    return hashlib.sha256(data).hexdigest()


def fingerprint(value):
    return sha(json.dumps(value, sort_keys=True, separators=(',', ':')).encode())


def periodic_span(tokens, period):
    best = run = end = 0
    for i in range(period, len(tokens)):
        run = run + 1 if tokens[i] == tokens[i - period] else 0
        if run >= 3 * period and run + period > best:
            best, end = run + period, i + 1
    return {'period_tokens': period, 'start_inclusive': end - best,
            'end_exclusive': end, 'span_tokens': best}


def atomic_json(path, value):
    temporary = None
    try:
        with tempfile.NamedTemporaryFile(mode='w', encoding='utf-8', dir=path.parent,
                                         prefix=path.name + '.', suffix='.tmp', delete=False) as stream:
            temporary = Path(stream.name)
            json.dump(value, stream, indent=2, allow_nan=False)
            stream.write('\n')
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)
    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)
